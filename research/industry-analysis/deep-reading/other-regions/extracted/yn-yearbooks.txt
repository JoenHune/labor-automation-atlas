function createPage(_nPageCount, _nCurrIndex, _sPageName, _sPageExt) {
    
    var page="";
    <!-- 总页数小于一页不显示 -->
    if (_nPageCount == null || _nPageCount <= 1) {
        return false;
    } 


    page += '<span class="number">共 '+28+' 条数据</span>';
    <!--  当前页是首页  -->
    if (_nCurrIndex == 0){
        page += '<a class="pageprv disabled" href="javascript:;" target="_self"><</a>';
        page += '<a class="mobile pageprv disabled" href="javascript:;" target="_self">上一页</a>';
        page += '<span class="mobile current-show">'+'1/'+_nPageCount+'</span>';
    }
    <!-- 上一页 -->
    if (_nCurrIndex > 0){
        if (_nCurrIndex == 1){
            var prePage = _sPageName + "." + _sPageExt;
        } else {
            var prePage = _sPageName + "_" + (_nCurrIndex - 1) + "." + _sPageExt;
        }
        page += '<a class="pageprv" href="' + prePage + '" target="_self"><</a>';
        page += '<a class="mobile pageprv" href="'+prePage+'" target="_self">上一页</a>';
        page += '<span class="mobile current-show">'+(_nCurrIndex+1)+'/'+_nPageCount+'</span>';
    }
    
    <!-- 前面大于三页 -->
    if (_nCurrIndex > 2) {
        page+='<a class="" href="' + _sPageName + "." + _sPageExt  + '" target="_self">1</a>'; 
    }
    
    <!-- 前面大于四页省略 -->
    if (_nCurrIndex > 3) {
        page+='<a target="_self">...</a>';
    }
    
    <!-- 列表 -->
    for(var i = 0; i < _nPageCount; i++){
        var sPageName;
        <!-- 前两页 -->
        if ((_nCurrIndex - 2) >= 0) { 
            if (i == (_nCurrIndex - 2)){
             if (i == 0) {
                 sPageName = _sPageName + "." + _sPageExt;
             } else {
                 sPageName = _sPageName + "_" + (_nCurrIndex - 2) + "." + _sPageExt;
             }
             page+='<a class="" href="' + sPageName + '" target="_self">' + ( i + 1 ) +'</a>';
            } 
        }
        if ((_nCurrIndex - 1) >= 0) {
             if (i == (_nCurrIndex - 1)){
                 if (i == 0) {
                     sPageName = _sPageName + "." + _sPageExt;
                 } else {
                     sPageName = _sPageName + "_" + (_nCurrIndex - 1) + "." + _sPageExt;
                 }
                 page+='<a class="" href="' + sPageName + '" target="_self">' + ( i + 1 ) +'</a>';
             } 
        }
        
        <!-- 当前页 -->
        if(i == _nCurrIndex){
            page+='<a class="cur">' + (i + 1) + '</a>';
        }
        
        <!-- 后两页 -->
        if ((_nCurrIndex + 1) <= (_nPageCount - 1)) { 
             if (i == (_nCurrIndex + 1)){                               
                 sPageName = _sPageName + "_" + (_nCurrIndex + 1) + "." + _sPageExt;
                 page+='<a class="" href="' + sPageName + '" target="_self">' + ( i + 1 ) +'</a>';
             } 
        }
        if ((_nCurrIndex + 2) <= (_nPageCount - 1)) {
             if (i == (_nCurrIndex + 2)){
                 sPageName = _sPageName + "_" + (_nCurrIndex + 2) + "." + _sPageExt;
                 page+='<a class="" href="' + sPageName + '" target="_self">' + ( i + 1 ) +'</a>';
             } 
        }  
    }

    for(var j = _nCurrIndex+3 ;j<( _nPageCount>5?5:_nPageCount); j++){
         sPageName = _sPageName + "_" + j + "." + _sPageExt;
     page+='<a class="" href="' + sPageName + '" target="_self">' + (j+1) +'</a>';
                
    }


        
    <!-- 后面大于四页省略 -->
    if (_nCurrIndex < (_nPageCount - 4)&&_nPageCount>6) {
        page+='<a>...</a>';
    }
    
    <!-- 后面大于三页 -->
    if (_nCurrIndex <_nPageCount-3&&_nPageCount>5) {
        page+='<a class="" href="' + _sPageName + "_" + (_nPageCount - 1) + "." + _sPageExt  + '" target="_self">'+_nPageCount+'</a>'; 
    }

    <!-- 下一页 -->
    if (_nCurrIndex < (_nPageCount - 1)){
        var prePage = _sPageName + "_" + (_nCurrIndex + 1) + "." + _sPageExt;
        page+='<a class="pagenext" href="' + prePage + '" target="_self">></a>';
        page += '<a class="mobile pagenext" href="'+prePage+'" target="_self">下一页</a>';
    }

    <!-- 当前页是尾页 -->
    if (_nCurrIndex == (_nPageCount - 1)){
        page+='<a class="pagenext disabled" href="javascript:;" target="_self">></a>';
        page += '<a class="mobile pagenext disabled" href="javascript:;" target="_self">下一页</a>';
    }

    
    <!-- 打印页码 -->
    $("#page").html(page);
}
