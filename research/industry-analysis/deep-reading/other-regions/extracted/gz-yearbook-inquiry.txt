function SetIsArchive() {
    let DocIsArchive = ($("meta[name='DocIsArchive']").length > 0) ? ($("meta[name='DocIsArchive']").attr("content").split("|")) : [0, 0];
    let ChannelIsArchive = ($("meta[name='ChannelIsArchive']").length > 0) ? ($("meta[name='ChannelIsArchive']").attr("content").split("|")) : [0, 0];
    if (parseInt(DocIsArchive[0]) == 1) {
        AddArchiveHtml(DocIsArchive[1]);
    } else if (parseInt(ChannelIsArchive[0]) == 1) {
        //AddArchiveHtml(ChannelIsArchive[1]);
        if (ChannelIsArchive[1] && ChannelIsArchive[1].indexOf('归档时间【') > -1) {
            let gdsj = ChannelIsArchive[1].substring(ChannelIsArchive[1].indexOf('归档时间【') + 5, ChannelIsArchive[1].indexOf('】'));
            AddArchiveHtml(gdsj);
        } else {
            AddArchiveHtml(ChannelIsArchive[2]);
        }
    }
}

            function getMyDate(timestamp) {
                var date = new Date(timestamp);//时间戳为10位需*1000，时间戳为13位的话不需乘1000
                Y = date.getFullYear() + '-';
                M = (date.getMonth()+1 < 10 ? '0' + (date.getMonth()+1) : date.getMonth()+1) + '-';
                D = (date.getDate() < 10 ? '0' + (date.getDate()) : date.getDate()) + ' ';
                h = (date.getHours() < 10 ? '0' + (date.getHours()) : date.getHours()) + ':';
                m = (date.getMinutes()< 10 ? '0' + (date.getMinutes()) : date.getMinutes()) + ':';
                s  = (date.getSeconds()< 10 ? '0' + (date.getSeconds()) : date.getSeconds());
                return Y + M + D + h + m + s;
            }
            function getdate() {
                var now = new Date(),
                        y = now.getFullYear(),
                        m = ("0" + (now.getMonth() + 1)).slice(-2),
                        d = ("0" + now.getDate()).slice(-2);
                return y + "-" + m + "-" + d + " " + now.toTimeString().substr(0, 8);
            }

            function GetStatus(status) {
                if (status == "4") {
                    status = "待办理";
                } else if (status == "5") {
                    status = "待审核";
                } else if (status == "7") {
                    status = "已办结";
                } else {
                    status = "办理中";
                }
                return status;
            }
            function GetScoreValue(status) {
                if (status == "1") {
                    status = "不满意";
                } else if (status == "2") {
                    status = "一般";
                } else if (status == "3") {
                    status = "基本满意";
                } else if (status == "4") {
                    status = "满意";
                } else if (status == "5") {
                    status = "非常满意";
                } else {
                    status = "暂未评价";
                }

                return status;
