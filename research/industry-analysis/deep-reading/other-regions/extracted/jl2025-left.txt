function setBackground(flag) {
		var obj = event.srcElement;
		if (obj == null)
			return false;
		var tabs = obj.parentNode.parentNode;
		var rowno = obj.parentNode.rowIndex + 2;
		if (tabs.rows[rowno].style.display == ""){
		    if (flag && event.srcElement == obj) {
			  obj.background = "./images/other/menubg_1.gif";
			} else {
		obj.background = "./images/other/menubg_.gif";
			}
		}
		else {
		    if (flag && event.srcElement == obj) {
			  obj.background = "./images/other/menubg1.gif";
			} else {
		obj.background = "./images/other/menubg.gif";
			}
		}
}
function   changeActiveRow(obj)    {   
	  if(currentActiveRow != null)   {
	    currentActiveRow.style.backgroundColor="";   
		currentActiveRow.parentNode.cells[0].style.backgroundColor="";
	  }
	  currentActiveRow=obj;   
	  obj.style.backgroundColor="#FAF1D4";   
	  obj.parentNode.cells[0].style.backgroundColor="#FBC334";
	  parent.window.frames["topFrame"].setActiveMenu(null, false);
  } 
 
function turnit(obj, cname){
	var rowno = obj.parentNode.rowIndex;
	var tables = obj.parentNode.parentNode;
	for (i = rowno + 2; i < tables.rows.length; i += 2)
		if (tables.rows[i].id.length != 0 && tables.rows[i].id == cname)
			break;
	rowno = i;
	for (i = 0; i < tables.rows.length; i += 1){
		if (tables.rows[i].id.length != 0 && tables.rows[i].id == cname){
			if (i == rowno){
				if (tables.rows[rowno].style.display == ""){
					tables.rows[i].style.display = "none";
				}
				else{
					tables.rows[i].style.display = "";
				}
			}
		}
	}
	obj.onClick = "turnit(obj, 'Content')";
}

function clickHandler() {
  var targetId, srcElement, targetElement;
  srcElement = window.event.srcElement;
  if (srcElement.className == "Outline") {
     targetId = srcElement.id + "details";
     targetElement = document.all(targetId);
     if (targetElement.style.display == "none") {
  targetElement.style.display = "";
     } else {
  targetElement.style.display = "none";
     }
  }
}

document.onclick = clickHandler;
