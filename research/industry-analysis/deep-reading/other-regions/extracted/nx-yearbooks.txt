function publish(pbIndexList,cn){
	//获取出版物的分组
	$.each(pbIndexList,function(i,pbIndex){
		var moreHref = pbIndex.morehref,
			code	 = pbIndex.code,
			html = '<div class="publicationL">'+
					'<img src="/files_nx_pub/pics/publish/'+pbIndex.picfile+'.jpg" width="150" height="212"></div>'+
					'<div class="publicationR"><h2 class="f14 fb pl10 pr">'+pbIndex.ctitle;
		
		html += moreHref == "" ? '</h2>' : (moreHref.indexOf("publish.htm?m=getMorePublish")>0 ? '<a href="'+pbIndex.morehref+'" class="more pa">更多 >></a></h2>' : '<a href="'+pbIndex.morehref+'" class="more pa" target="_blank">更多 >></a></h2>');
		
		html += '<div class="publicationText t2 p15">'+pbIndex.cexp+'</div>';
				
		//获取出版物每组的年份
		var index="'"+pbIndex.ctitle+"'";
		if(pbIndex.isgroup == "1"){
			html += '<ul class="pl10 pr10 clearfix">';
			$.each(pbIndex.publishGroupList,function(i,data){
				$.each(data.publishList,function(i,list){
					var hrf="'"+list.href+"'";
					var ctitle="'"+list.ctitle+"'";
					html += '<li><div class="publishline"><a href="javascript:getPublish(' + hrf + ','+index+','+ctitle+',true)">'+ list.ctitle +'</a></div></li>';
				});
			});

			html += '</ul>';
		}
		else{
			html += '<ul class="pl10 pr10 clearfix">';
			
			$.each(pbIndex.publishGroupList,function(i,data){
				$.each(data.publishList,function(i,list){
					var hrf="'"+list.href+"'";
					var ctitle="'"+list.ctitle+"'";
					
					html += '<li><div class="publishline"><a href="javascript:getPublish(' + hrf + ','+index+','+ctitle+',true)">'+ list.ctitle +'</a></div></li>';
				});
			});
			html += '</ul>';
		}
		
		//var IEInfo = '';
		/* if(pbIndex.ctitle.indexOf("年鉴") != -1){
			//html += '<div class="left ieinfo">注：建议使用IE7以上版本普通模式(非“兼容性视图”)，或360浏览器极速模式浏览。</div>';
		} */
		html += '</div>';
		$(".publication").append('<div class="publicationList pr pb20">' + html + '</div>');
	});
};

	function search() {
		var str = $("#keyword").val();
		if ($.trim(str) != "" && str != "如：2014年 宁夏 CPI") {
			window.location = "/search.htm?s=" + str;
		}
	}
	$("#keyword").focus(function() {
		$(this).val() == "如：2014年 宁夏 CPI" && $(this).val("");
	}).blur(function() {
		$.trim($("#keyword").val()) == "" && $(this).val("如：2014年 宁夏 CPI");
	}).keydown(function(e) {
		e.keyCode == 13 && search();
	});
