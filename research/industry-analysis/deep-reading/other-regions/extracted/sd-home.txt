function fixImage1(ImgD, width, height) {
        var image = new Image();
        image.src = ImgD.src;
        imgw = image.width;
        imgh = image.height;
        if (imgw * 1 > width) {
            imgh = imgh / imgw * width;
            imgw = width;
        }
        if (imgh * 1 > height) {
            imgw = imgw / imgh * height;
            imgh = height;
        }
        ImgD.width = imgw;
        ImgD.height = imgh;
    }

var LTPspeed=20;//数值越大，速度越慢 
var LTP2=document.getElementById("LTP2");
var LTP1=document.getElementById("LTP1");
var LTP=document.getElementById("LTP");
LTP2.innerHTML=LTP1.innerHTML;
LTP.scrollLeft=LTP.scrollWidth
function LTPMarqueeLeft(){
if(LTP2.offsetWidth-LTP.scrollLeft<=0)
LTP.scrollLeft-=LTP1.offsetWidth
else{
LTP.scrollLeft++
}
}
function LTPMarqueeRight(){
if(LTP.scrollLeft<=0)
LTP.scrollLeft+=LTP2.offsetWidth
else{
LTP.scrollLeft--
}
}
var LTPMar=setInterval(LTPMarqueeRight,LTPspeed);
LTP.onmouseover=function() {clearInterval(LTPMar);}
LTP.onmouseout=function() {LTPMar=setInterval(LTPMarqueeRight,LTPspeed);}

function changeRedirect(LorR,PlusSpeed)
{
 clearInterval(LTPMar);
 if(LorR==0) 
 {
  LTPMar=setInterval(LTPMarqueeLeft,PlusSpeed); 
  LTP.onmouseout=function() {LTPMar=setInterval(LTPMarqueeLeft,PlusSpeed);}
 }
 else
 {
  LTPMar=setInterval(LTPMarqueeRight,PlusSpeed);
  LTP.onmouseout=function() {LTPMar=setInterval(LTPMarqueeRight,PlusSpeed);}
 }
}
