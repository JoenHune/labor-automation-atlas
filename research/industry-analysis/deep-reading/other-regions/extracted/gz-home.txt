        AddArchiveHtml(DocIsArchive[1]);

        //AddArchiveHtml(ChannelIsArchive[1]);

            AddArchiveHtml(gdsj);
        } else {
            AddArchiveHtml(ChannelIsArchive[2]);
        }
    }
}
