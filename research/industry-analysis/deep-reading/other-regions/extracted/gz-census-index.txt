function SetIsArchive() {
    let DocIsArchive = ($("meta[name='DocIsArchive']").length > 0) ? ($("meta[name='DocIsArchive']").attr("content").split("|")) : [0, 0];
    let ChannelIsArchive = ($("meta[name='ChannelIsArchive']").length > 0) ? ($("meta[name='ChannelIsArchive']").attr("content").split("|")) : [0, 0];
    if (parseInt(DocIsArchive[0]) == 1) {
        AddArchiveHtml(DocIsArchive[1]);
    } else if (parseInt(ChannelIsArchive[0]) == 1) {
        //AddArchiveHtml(ChannelIsArchive[1]);
        if (ChannelIsArchive[1] && ChannelIsArchive[1].indexOf('归档时间【') > -1) {
            let gdsj = ChannelIsArchive[1].substring(ChannelIsArchive[1].indexOf('归档时间【') + 5, ChannelIsArchive[1].indexOf('】'));
            AddArchiveHtml(gdsj);
        } else {
            AddArchiveHtml(ChannelIsArchive[2]);
        }
    }
}
